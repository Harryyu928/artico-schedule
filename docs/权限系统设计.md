### 2.3 飞书配置项

```typescript
interface FeishuConfig {
  // 飞书应用配置
  appId: string;
  appSecret: string;
  
  // 多维表格配置
  bitable: {
    appToken: string;
    tables: {
      students: string;      // 学生表ID
      teachers: string;      // 导师表ID
      selectionForms: string; // 选课单表ID
      classRecords: string;  // 上课记录表ID
      applications: string;  // 申请院校表ID
    };
  };
  
  // 日历配置
  calendar: {
    calendarId: string;
    defaultReminders: number[]; // 默认提醒时间（分钟）
  };
  
  // 通知配置
  notification: {
    enabled: boolean;
    channels: {
      scheduleConfirm: boolean;  // 排课确认通知
      scheduleChange: boolean;   // 排课变更通知
      homeworkReminder: boolean; // 作业提醒
      progressReport: boolean;   // 进度报告
    };
  };
  
  // 同步配置
  sync: {
    enabled: boolean;
    interval: number; // 同步间隔（分钟）
    direction: 'bidirectional' | 'to_feishu' | 'from_feishu';
  };
}
```

## 4. 数据权限过滤

### 4.1 数据访问规则

```typescript
// 数据过滤中间件伪代码
const dataFilter = {
  // 学生数据过滤
  student: {
    admin: () => true,              // 全部
    consultant: () => true,          // 全部
    teacher: (student, user) => 
      student.assignedTeachers.includes(user.teacherId), // 仅自己的学生
  },
  
  // 排课数据过滤
  schedule: {
    admin: () => true,
    consultant: () => true,
    teacher: (schedule, user) => 
      schedule.teacherId === user.teacherId, // 仅自己的排课
  },
  
  // 上课记录过滤
  classRecord: {
    admin: () => true,
    consultant: () => true,
    teacher: (record, user) => 
      record.teacherId === user.teacherId, // 仅自己的记录
  },
};
```

### 5.2 用户匹配规则

```typescript
// 飞书用户匹配逻辑
async function matchFeishuUser(feishuUserInfo: FeishuUserInfo) {
  // 1. 尝试通过飞书open_id匹配
  const existingUser = await db.query.users.findFirst({
    where: eq(users.feishuOpenId, feishuUserInfo.open_id)
  });
  
  if (existingUser) {
    return existingUser;
  }
  
  // 2. 尝试通过邮箱匹配（全职导师）
  const teacherByEmail = await db.query.teachers.findFirst({
    where: eq(teachers.email, feishuUserInfo.email)
  });
  
  if (teacherByEmail) {
    // 创建用户并关联
    return await createUserWithFeishu(teacherByEmail, feishuUserInfo);
  }
  
  // 3. 未匹配到，提示联系管理员
  throw new Error('用户未在系统中注册，请联系管理员');
}
```
